===== USER MODEL =====
"""User identity persistence model."""

from datetime import datetime
from typing import TYPE_CHECKING
from uuid import UUID, uuid4

from sqlalchemy import CheckConstraint, DateTime, String, Uuid, text
from sqlalchemy.orm import Mapped, mapped_column, relationship, validates

if TYPE_CHECKING:
    from agentforge.domain.models.workspace import Workspace, WorkspaceMembership

from agentforge.database.base import Base
from agentforge.domain.models.mixins import TimestampMixin


class User(TimestampMixin, Base):
    """A platform user, with authentication fields kept separate from operator secrets."""

    __tablename__ = "users"
    __table_args__ = (
        CheckConstraint(
            "account_status IN ('pending', 'active', 'suspended', 'deleted')",
            name="ck_users_account_status",
        ),
        CheckConstraint("email = lower(btrim(email))", name="ck_users_normalized_email"),
    )

    id: Mapped[UUID] = mapped_column(
        Uuid(as_uuid=True),
        primary_key=True,
        default=uuid4,
        server_default=text("gen_random_uuid()"),
    )
    email: Mapped[str] = mapped_column(String(320), nullable=False, unique=True, index=True)
    password_hash: Mapped[str | None] = mapped_column(String(255))
    account_status: Mapped[str] = mapped_column(
        String(20),
        nullable=False,
        default="pending",
        server_default=text("'pending'"),
    )
    email_verified_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True))
    last_login_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True))

    owned_workspaces: Mapped[list["Workspace"]] = relationship(
        back_populates="owner",
        foreign_keys="Workspace.owner_id",
    )
    workspace_memberships: Mapped[list["WorkspaceMembership"]] = relationship(
        back_populates="user",
        cascade="all, delete-orphan",
        passive_deletes=True,
    )

    @validates("email")
    def normalize_email(self, _key: str, value: str) -> str:
        return value.strip().lower()
===== WORKFLOW MODEL =====
"""Versioned workflow persistence model."""

from typing import TYPE_CHECKING, Any
from uuid import UUID, uuid4

from sqlalchemy import (
    CheckConstraint,
    ForeignKey,
    Integer,
    String,
    Text,
    UniqueConstraint,
    Uuid,
    text,
)
from sqlalchemy.dialects.postgresql import JSONB
from sqlalchemy.orm import Mapped, mapped_column, relationship

from agentforge.database.base import Base
from agentforge.domain.models.mixins import TimestampMixin

if TYPE_CHECKING:
    from agentforge.domain.models.execution import WorkflowExecution
    from agentforge.domain.models.user import User
    from agentforge.domain.models.workspace import Workspace


class Workflow(TimestampMixin, Base):
    """A named, validated workflow definition."""

    __tablename__ = "workflows"
    __table_args__ = (
        UniqueConstraint(
            "workspace_id",
            "name",
            "version",
            name="uq_workflows_workspace_name_version",
        ),
        UniqueConstraint("workspace_id", "id", name="uq_workflows_workspace_id"),
        CheckConstraint("version > 0", name="ck_workflows_positive_version"),
    )

    id: Mapped[UUID] = mapped_column(
        Uuid(as_uuid=True),
        primary_key=True,
        default=uuid4,
        server_default=text("gen_random_uuid()"),
    )
    name: Mapped[str] = mapped_column(String(200), nullable=False, index=True)
    description: Mapped[str | None] = mapped_column(Text)
    workflow_definition: Mapped[dict[str, Any]] = mapped_column(
        JSONB,
        nullable=False,
        server_default=text("'{}'::jsonb"),
    )
    version: Mapped[int] = mapped_column(Integer, nullable=False, default=1)
    workspace_id: Mapped[UUID] = mapped_column(
        Uuid(as_uuid=True),
        ForeignKey("workspaces.id", ondelete="RESTRICT"),
        nullable=False,
        index=True,
    )
    created_by_user_id: Mapped[UUID | None] = mapped_column(
        Uuid(as_uuid=True),
        ForeignKey("users.id", ondelete="SET NULL"),
        index=True,
    )
    updated_by_user_id: Mapped[UUID | None] = mapped_column(
        Uuid(as_uuid=True),
        ForeignKey("users.id", ondelete="SET NULL"),
        index=True,
    )

    workspace: Mapped["Workspace"] = relationship(back_populates="workflows")
    created_by: Mapped["User | None"] = relationship(
        foreign_keys=[created_by_user_id],
    )
    updated_by: Mapped["User | None"] = relationship(
        foreign_keys=[updated_by_user_id],
    )
    executions: Mapped[list["WorkflowExecution"]] = relationship(
        back_populates="workflow",
        foreign_keys="WorkflowExecution.workflow_id",
        passive_deletes="all",
        overlaps="workspace",
    )

class WorkflowExecution(Base):
    """A durable execution, including the state captured before planning."""

    __tablename__ = "workflow_executions"
    __table_args__ = (
        ForeignKeyConstraint(
            ["workspace_id", "workflow_id"],
            ["workflows.workspace_id", "workflows.id"],
            ondelete="RESTRICT",
            name="fk_workflow_executions_workspace_workflow",
        ),
        CheckConstraint(
            "status IN ('submitted', 'planning', 'planned', 'queued', 'running', "
            "'waiting_approval', 'succeeded', 'failed', 'cancelled', 'pending', 'completed')",
            name="ck_workflow_executions_status",
        ),
        CheckConstraint(
            "plan_version IS NULL OR plan_version > 0",
            name="ck_execution_plan_version",
        ),
        CheckConstraint(
            "status NOT IN ('succeeded', 'failed', 'cancelled', 'completed') "
            "OR completed_at IS NOT NULL",
            name="ck_terminal_execution_has_completion_time",
        ),
    )

    id: Mapped[UUID] = mapped_column(
        Uuid(as_uuid=True),
        primary_key=True,
        default=uuid4,
        server_default=text("gen_random_uuid()"),
    )
    workflow_id: Mapped[UUID | None] = mapped_column(
        Uuid(as_uuid=True),
        ForeignKey("workflows.id", ondelete="RESTRICT"),
        index=True,
    )
    workspace_id: Mapped[UUID] = mapped_column(
        Uuid(as_uuid=True),
        ForeignKey("workspaces.id", ondelete="RESTRICT"),
        nullable=False,
        index=True,
    )
    created_by_user_id: Mapped[UUID | None] = mapped_column(
        Uuid(as_uuid=True),
        ForeignKey("users.id", ondelete="SET NULL"),
        index=True,
    )
    started_by_user_id: Mapped[UUID | None] = mapped_column(
        Uuid(as_uuid=True),
        ForeignKey("users.id", ondelete="SET NULL"),
        index=True,
    )
    cancelled_by_user_id: Mapped[UUID | None] = mapped_column(
        Uuid(as_uuid=True),
        ForeignKey("users.id", ondelete="SET NULL"),
        index=True,
    )
    original_task: Mapped[str] = mapped_column(Text, nullable=False)
    status: Mapped[str] = mapped_column(String(30), nullable=False, default="submitted")
    plan_version: Mapped[int | None] = mapped_column(Integer)
    provider_mode: Mapped[str | None] = mapped_column(String(50))
    terminal_reason: Mapped[str | None] = mapped_column(Text)
    report: Mapped[dict[str, Any] | None] = mapped_column(JSONB)
    started_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True))
    completed_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True))
    created_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        nullable=False,
        server_default=func.now(),
    )

    workspace: Mapped["Workspace"] = relationship(
        back_populates="executions",
        foreign_keys=[workspace_id],
    )
    created_by: Mapped["User | None"] = relationship(
        foreign_keys=[created_by_user_id],
    )
    started_by: Mapped["User | None"] = relationship(
        foreign_keys=[started_by_user_id],
    )
    cancelled_by: Mapped["User | None"] = relationship(
        foreign_keys=[cancelled_by_user_id],
    )
    workflow: Mapped["Workflow | None"] = relationship(
        back_populates="executions",
        foreign_keys=[workflow_id],
    )
    steps: Mapped[list["ExecutionStep"]] = relationship(
        back_populates="execution",
        cascade="all, delete-orphan",
        passive_deletes=True,
    )
    approval_requests: Mapped[list["ApprovalRequest"]] = relationship(
        back_populates="execution",
        cascade="all, delete-orphan",
        passive_deletes=True,
    )
    tool_execution_records: Mapped[list["ToolExecutionRecord"]] = relationship(
        back_populates="execution",
        cascade="all, delete-orphan",
        passive_deletes=True,
    )
    events: Mapped[list["ExecutionEvent"]] = relationship(
        back_populates="execution",
        passive_deletes="all",
        order_by="ExecutionEvent.sequence_number",
    )
